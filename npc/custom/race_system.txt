//===== rAthena Script ========================================
//= Race System (Thai) - rA-E custom
//===== Description: ========================================
//= Inspired by the Roblox "Races" system.
//=  - 10 races with different roll chances, 3 star levels each
//=  - 6 race slots: slots 1-2 are free, slots 3-6 are opened with Race Slot Key
//=  - a Race Roll item re-rolls ONE chosen slot (open and not locked)
//=  - rolling a race you already own raises its stars (shared by all slots)
//=  - ONE active slot gives the bonuses, hat effect and status icon
//===== Items: ========================================
//= 32501 Race_Roll      - one roll of all unlocked slots
//= 32502 Race_Slot_Key  - opens the next extra slot (3..6)
//= 32503 Race_Book      - opens this menu (also: @race)
//===== Notes: ========================================
//= ** SAVE AS TIS-620 / cp874 ** (Thai client)
//= Bonuses are applied by the status SC_QUEST_BUFF3 (db/pre-re/status.yml)
//= which calls function F_RaceBonus below.
//= Permanent char variables: race_s1..6 (race in slot), race_lock (bitmask),
//= race_open (extra slots opened 0..4), race_act (active slot 1..6),
//= race_st1..10 (stars of each race 0..3)
//========================================

-	script	RaceSystem	-1,{
	end;

OnInit:
	bindatcmd "race", "RaceSystem::OnOpen";
	end;

OnOpen:
	callfunc "F_RaceUI";
	end;

OnOpenItem:
	callfunc "F_RaceUI";
	end;

OnPCLoginEvent:
	callfunc "F_RaceApply";
	end;
}

// F_RaceRarity(tier, 0 = name | 1 = color)
function	script	F_RaceRarity	{
	switch (getarg(0)) {
	case 1: return getarg(1) ? "^6B7280" : "สามัญ";
	case 2: return getarg(1) ? "^15803D" : "ไม่ธรรมดา";
	case 3: return getarg(1) ? "^1D4ED8" : "หายาก";
	case 4: return getarg(1) ? "^7E22CE" : "เอพิค";
	}
	return getarg(1) ? "^C2410C" : "ตำนาน";
}

// Colored star bar: F_RaceStarBar(stars) -> [***] / [*--] / [---]
function	script	F_RaceStarBar	{
	.@n = getarg(0);
	.@t$ = "^D97706";
	for (.@i = 1; .@i <= 3; .@i++) {
		if (.@i == .@n + 1)
			.@t$ += "^C4C4C4";
		.@t$ += (.@i <= .@n) ? "*" : "-";
	}
	return "[" + .@t$ + "^000000]";
}

// Tier used only to colour the bullet of a perk line (1 EXP, 2 Drop, 3 DEF/Speed/Luk, 4 Crit/Skill damage, 5 Skill CD).

// Perk lines of a race. F_RacePerkLines(race, star {, compact})
// Full mode: one line per perk with the 3 star levels, current star in green; the bullet is colored by the tier that introduced the perk.
// Compact mode (3rd arg 1): current star value only, two perks per line. Lines are joined with "|".
function	script	F_RacePerkLines	{
	.@r = getarg(0);
	.@st = getarg(1);
	.@cmp = getarg(2, 0);
	.@foc$ = "," + F_RaceData(.@r, 6) + ",";
	.@rc$ = F_RaceData(.@r, 2);
	deletearray .@perk$[0], getarraysize(.@perk$);
	explode(.@perk$, F_RaceData(.@r, 4), ",");
	.@np = getarraysize(.@perk$);
	.@out$ = "";
	for (.@i = 0; .@i < .@np; .@i++) {
		deletearray .@v$[0], getarraysize(.@v$);
		explode(.@v$, .@perk$[.@i], ":");
		.@ty = atoi(.@v$[0]);
		.@u$ = "%";
		.@bul$ = F_RaceRarity(F_RacePerkTier(.@ty), 1) + "+^000000 ";
		.@mk$ = (compare(.@foc$, "," + .@ty + ",")) ? " ^C2410C(เด่น)^000000" : "";
		if (.@cmp) {
			.@cur = .@st < 1 ? 1 : .@st;
			.@t$ = .@bul$ + F_RacePerkName(.@ty) + " " + .@rc$ + "+" + .@v$[.@cur] + .@u$ + "^000000" + .@mk$;
			if (.@i % 2 == 0 && .@i < .@np - 1)
				.@out$ += .@t$ + "   ";
			else
				.@out$ += .@t$ + (.@i < .@np - 1 ? "|" : "");
		} else {
			.@t$ = .@bul$ + F_RacePerkName(.@ty) + .@mk$ + "  ";
			for (.@j = 1; .@j <= 3; .@j++) {
				.@t$ += (.@j == .@st ? .@rc$ : "^9CA3AF") + "+" + .@v$[.@j] + .@u$ + "^000000";
				if (.@j < 3)
					.@t$ += " / ";
			}
			.@out$ += .@t$ + (.@i < .@np - 1 ? "|" : "");
		}
	}
	return .@out$;
}

// Applies the bonuses of race getarg(0). Called from the status script (SC_QUEST_BUFF3) so bonus commands work.
function	script	F_RaceBonus	{
	.@r = getarg(0);
	if (.@r < 1 || .@r > 10)
		return;
	.@st = getd("race_st" + .@r);
	if (.@st < 1)
		.@st = 1;
	if (.@st > 3)
		.@st = 3;
	deletearray .@perk$[0], getarraysize(.@perk$);
	explode(.@perk$, F_RaceData(.@r, 4), ",");
	.@n = getarraysize(.@perk$);
	for (.@i = 0; .@i < .@n; .@i++) {
		explode(.@v$, .@perk$[.@i], ":");
		.@val = atoi(.@v$[.@st]);
		switch (atoi(.@v$[0])) {
		case 1:  bonus2 bExpAddRace, RC_All, .@val; break;
		case 2:  bonus2 bDropAddRace, RC_All, .@val; break;
		case 3:  bonus bDelayrate, -.@val; break;
		case 6:  bonus bDefRate, .@val; break;
		case 7:  break; // move speed is applied in C++ (status_calc_speed) from SC_QUEST_BUFF3 val2: it stacks like a mount
		case 8:  bonus bCritical, .@val; break;
		case 12: bonus bAspdRate, .@val; break;
		case 14: break; // Luk% is applied in C++ (status_calc_luk) from SC_QUEST_BUFF3 val3
		case 15: break; // Skill damage% is applied in C++ (battle_skill_damage) from SC_QUEST_BUFF3 val4
		}
	}
	return;
}

// Value of perk type getarg(1) of race getarg(0) at its current star (0 when the race has no such perk)
function	script	F_RacePerkVal	{
	.@r = getarg(0);
	.@st = getd("race_st" + .@r);
	if (.@st < 1)
		.@st = 1;
	if (.@st > 3)
		.@st = 3;
	deletearray .@perk$[0], getarraysize(.@perk$);
	explode(.@perk$, F_RaceData(.@r, 4), ",");
	for (.@i = 0; .@i < getarraysize(.@perk$); .@i++) {
		deletearray .@v$[0], getarraysize(.@v$);
		explode(.@v$, .@perk$[.@i], ":");
		if (atoi(.@v$[0]) == getarg(1))
			return atoi(.@v$[.@st]);
	}
	return 0;
}

// (Re)applies the active race: status (bonus + icon) and hat effects.
// Hat effects: ALL effects of the race are shown at every star level (others identify the race by its effects). Only effects whose .str exists in the client
// (checked with check_hat_effects.py) may be listed in F_RaceData field 3.
function	script	F_RaceApply	{
	.@a = race_act;
	.@r = 0;
	if (.@a >= 1 && .@a <= 6)
		.@r = getd("race_s" + .@a);
	sc_end SC_QUEST_BUFF3;
	// pass 1: everything off (some races share an effect), pass 2: the active race on
	for (.@i = 1; .@i <= 10; .@i++) {
		deletearray .@e$[0], getarraysize(.@e$);
		explode(.@e$, F_RaceData(.@i, 3), ",");
		for (.@j = 0; .@j < getarraysize(.@e$); .@j++)
			if (.@e$[.@j] != "")
				hateffect getd(.@e$[.@j]), 0;
	}
	if (.@r) {
		deletearray .@e$[0], getarraysize(.@e$);
		explode(.@e$, F_RaceData(.@r, 3), ",");
		for (.@j = 0; .@j < getarraysize(.@e$); .@j++)
			if (.@e$[.@j] != "")
				hateffect getd(.@e$[.@j]), 1;
	}
	if (.@r) {
		// val1 = race + stars * 100 (the stars are shown after the character name by the client packet code)
		.@sv = getd("race_st" + .@r);
		if (.@sv < 1) .@sv = 1;
		if (.@sv > 3) .@sv = 3;
		sc_start4 SC_QUEST_BUFF3, -1, .@r + .@sv * 100, F_RacePerkVal(.@r, 7), F_RacePerkVal(.@r, 14), F_RacePerkVal(.@r, 15);
	}
	return;
}

function	script	F_RaceUI	{
	if (race_inited == 0) {
		race_inited = 1;
		race_act = 0;
	}
	while (1) {
		.@avail = 2 + race_open;
		callsub L_Title, "ระบบเผ่า";
		callsub L_Slots, .@avail;
		.@ar = (race_act >= 1 && race_act <= 6) ? getd("race_s" + race_act) : 0;
		if (.@ar) {
			.@rc$ = F_RaceData(.@ar, 2);
			mes .@rc$ + "========================================^000000";
			mes " " + F_RaceName(.@ar) + "   " + F_RaceStarBar(getd("race_st" + .@ar)) + "   " + F_RaceRarity(F_RaceData(.@ar, 5), 1) + F_RaceRarity(F_RaceData(.@ar, 5), 0) + "^000000   ^6B7280ช่อง " + race_act + "^000000";
			mes " ^6B7280แนวทาง:^000000 " + .@rc$ + F_RaceData(.@ar, 7) + "^000000";
			mes .@rc$ + "- - - - - - - - - - - - - - - - - - - - -^000000";
			deletearray .@ln$[0], getarraysize(.@ln$);
			explode(.@ln$, F_RacePerkLines(.@ar, getd("race_st" + .@ar), 1), "|");
			for (.@i = 0; .@i < getarraysize(.@ln$); .@i++)
				mes " " + .@ln$[.@i];
			mes .@rc$ + "========================================^000000";
		} else {
			mes "^9CA3AF ยังไม่ได้เลือกเผ่าที่ใช้งาน (เมนู 'เลือกเผ่าที่ใช้งาน')^000000";
		}
		mes " Race Roll: ^1E3A8A" + countitem(32501) + "^000000    Race Slot Key: ^1E3A8A" + countitem(32502) + "^000000";
		.@s = select("สุ่มเผ่าในช่อง (ใช้ Race Roll 1 ชิ้น)", "เลือกเผ่าที่ใช้งาน", "ล็อก / ปลดล็อกช่อง", "ปลดช่องเพิ่ม (Race Slot Key)", "ดูโบนัสของเผ่า", "ปิด");

		case 1: callsub L_Roll, .@avail; break;
		case 2: callsub L_Pick, .@avail; break;
		case 3: callsub L_Lock, .@avail; break;
		case 4: callsub L_Open; break;
		case 5: callsub L_Info; break;

			callsub L_Title, "ระบบเผ่า";
			mes "ปิดเมนูแล้ว พิมพ์ ^0000FF@race^000000 เพื่อเปิดอีกครั้ง";
			close;
		}
	}
	end;

L_Roll:

	if (countitem(32501) < 1) {
		callsub L_Msg, "ต้องมี Race Roll อย่างน้อย 1 ชิ้น";
		return;
	}
	callsub L_Title, "เลือกช่องที่จะสุ่ม";
	mes "สุ่มทีละ 1 ช่อง ใช้ Race Roll 1 ชิ้น";
	callsub L_Slots, .@avail;
	.@menu$ = "";
	for (.@i = 1; .@i <= 6; .@i++) {
		.@r = (.@i <= .@avail) ? getd("race_s" + .@i) : 0;
		if (.@i > .@avail)
			.@menu$ += "[" + .@i + "] (ยังไม่ปลดล็อก):";
		else if (race_lock & (1 << .@i))
			.@menu$ += "[" + .@i + "] (ล็อกอยู่):";
		else if (.@r)
			.@menu$ += "[" + .@i + "] " + F_RaceData(.@r, 0) + " -> สุ่มใหม่:";
		else
			.@menu$ += "[" + .@i + "] ว่าง -> สุ่ม:";
	}
	.@menu$ += "ยกเลิก";
	.@c = select(.@menu$);
	if (.@c > 6)
		return;
	if (.@c > .@avail) {
		callsub L_Msg, "ช่องนี้ยังไม่ได้ปลดล็อก";
		return;
	}
	if (race_lock & (1 << .@c)) {
		callsub L_Msg, "ช่องนี้ถูกล็อกอยู่ ปลดล็อกก่อนจึงจะสุ่มได้";
		return;
	}
	if (countitem(32501) < 1)
		return;
	delitem 32501, 1;
	.@x = rand(1000);
	.@acc = 0;
	.@pick = 1;
	for (.@r = 1; .@r <= 10; .@r++) {
		.@acc += F_RaceData(.@r, 1);
		if (.@x < .@acc) {
			.@pick = .@r;
			break;
		}
	}
	setd "race_s" + .@c, .@pick;
	// every roll of a race counts as one more star for that race (all slots holding it share the stars)
	.@old = getd("race_st" + .@pick);
	.@new = (.@old < 3) ? .@old + 1 : 3;
	setd "race_st" + .@pick, .@new;
	callsub L_Title, "ผลการสุ่มช่อง " + .@c;
	.@rc$ = F_RaceData(.@pick, 2);
	mes .@rc$ + "========================================^000000";
	mes " ได้รับเผ่า  " + F_RaceName(.@pick) + "   " + F_RaceStarBar(.@new) + "   " + F_RaceRarity(F_RaceData(.@pick, 5), 1) + F_RaceRarity(F_RaceData(.@pick, 5), 0) + "^000000";
	if (.@old == 0)
		mes " ^15803Dเผ่าใหม่!^000000 " + F_RaceStarTxt(.@new);
	else if (.@old < 3)
		mes " ^15803Dดาวเพิ่ม " + .@old + " -> " + .@new + "^000000 (ทุกช่องที่เป็นเผ่านี้อัพเดตด้วย)";
	else
		mes " ดาวเต็มแล้ว (3/3)";
	mes " ^6B7280แนวทาง:^000000 " + .@rc$ + F_RaceData(.@pick, 7) + "^000000";
	mes .@rc$ + "========================================^000000";
	callfunc "F_RaceApply";
	next;
	return;

L_Pick:
	.@avail = getarg(0);
	callsub L_Title, "เลือกเผ่าที่ใช้งาน";
	callsub L_Slots, .@avail;
	.@menu$ = "";
	for (.@i = 1; .@i <= 6; .@i++) {
		.@r = (.@i <= .@avail) ? getd("race_s" + .@i) : 0;
		if (.@r)
			.@menu$ += "[" + .@i + "] " + F_RaceData(.@r, 0) + " (" + getd("race_st" + .@r) + "/3)" + (.@i == race_act ? " *" : "") + ":";
		else
			.@menu$ += "[" + .@i + "] -:";
	}
	.@menu$ += "ไม่ใช้เผ่า (ปิดโบนัส):ยกเลิก";
	.@c = select(.@menu$);
	if (.@c == 8)
		return;
	if (.@c == 7) {
		race_act = 0;
	} else {
		if (.@c > .@avail || !getd("race_s" + .@c)) {
			callsub L_Msg, "ช่องนี้ยังไม่มีเผ่า";
			return;
		}
		race_act = .@c;
	}
	callfunc "F_RaceApply";
	callsub L_Msg, "เปลี่ยนเผ่าที่ใช้งานเรียบร้อย";
	return;

L_Lock:
	.@avail = getarg(0);
	callsub L_Title, "ล็อก / ปลดล็อกช่อง";
	mes "ช่องที่ล็อกจะไม่ถูกสุ่มทับ";
	callsub L_Slots, .@avail;
	.@menu$ = "";
	for (.@i = 1; .@i <= 6; .@i++) {
		if (.@i > .@avail)
			.@menu$ += "[" + .@i + "] (ยังไม่ปลดล็อก):";
		else
			.@menu$ += "[" + .@i + "] " + ((race_lock & (1 << .@i)) ? "ปลดล็อก" : "ล็อก") + ":";
	}
	.@menu$ += "ยกเลิก";
	.@c = select(.@menu$);
	if (.@c >= 7 || .@c > .@avail)
		return;
	race_lock = race_lock ^ (1 << .@c);
	return;

L_Open:
	if (race_open >= 4) {
		callsub L_Msg, "ปลดล็อกครบทุกช่องแล้ว";
		return;
	}
	if (countitem(32502) < 1) {
		callsub L_Msg, "ต้องมี Race Slot Key 1 ชิ้นเพื่อปลดช่องที่ " + (3 + race_open);
		return;
	}
	delitem 32502, 1;
	race_open++;
	callsub L_Msg, "ปลดล็อกช่องที่ " + (2 + race_open) + " แล้ว";
	return;

L_Info:
	callsub L_Title, "เลือกเผ่าที่ต้องการดู";
	for (.@r = 1; .@r <= 10; .@r++)
		mes " " + F_RaceData(.@r, 2) + "|||^000000 " + F_RaceName(.@r) + "  " + F_RaceRarity(F_RaceData(.@r, 5), 1) + F_RaceRarity(F_RaceData(.@r, 5), 0) + "^000000  ^6B7280" + (F_RaceData(.@r, 1) / 10) + "." + (F_RaceData(.@r, 1) % 10) + "%^000000";
	.@menu$ = "";
	for (.@r = 1; .@r <= 10; .@r++)
		.@menu$ += F_RaceData(.@r, 0) + " - " + F_RaceRarity(F_RaceData(.@r, 5), 0) + " (" + (F_RaceData(.@r, 1) / 10) + "." + (F_RaceData(.@r, 1) % 10) + "%):";
	.@menu$ += "ยกเลิก";
	.@r = select(.@menu$);
	if (.@r > 10)
		return;
	callsub L_Title, F_RaceData(.@r, 0);
	.@st = getd("race_st" + .@r);
	.@rc$ = F_RaceData(.@r, 2);
	mes .@rc$ + "========================================^000000";
	mes " " + F_RaceName(.@r) + "   " + F_RaceStarBar(.@st) + "   " + F_RaceRarity(F_RaceData(.@r, 5), 1) + F_RaceRarity(F_RaceData(.@r, 5), 0) + "^000000";
	mes " ^6B7280โอกาสสุ่มได้^000000 " + (F_RaceData(.@r, 1) / 10) + "." + (F_RaceData(.@r, 1) % 10) + "%   ^6B7280(เฉลี่ยประมาณ " + (1000 / F_RaceData(.@r, 1)) + " ครั้ง)^000000";
	mes " ^6B7280แนวทาง:^000000 " + .@rc$ + F_RaceData(.@r, 7) + "^000000";
	mes .@rc$ + "- - - - - - - - - - - - - - - - - - - - -^000000";
	mes "^9CA3AF ค่าเรียงเป็น ดาว 1 / ดาว 2 / ดาว 3  (สีเผ่า = ดาวปัจจุบัน)^000000";
	deletearray .@ln$[0], getarraysize(.@ln$);
	explode(.@ln$, F_RacePerkLines(.@r, .@st), "|");
	for (.@i = 0; .@i < getarraysize(.@ln$); .@i++)
		mes " " + .@ln$[.@i];
	mes .@rc$ + "========================================^000000";
	next;
	return;

L_Slots:
	.@avail = getarg(0);
	for (.@i = 1; .@i <= 6; .@i++) {
		.@r = getd("race_s" + .@i);
		.@mk$ = (.@i == race_act) ? "^15803D>>^000000" : "   ";
		if (.@i > .@avail) {
			mes .@mk$ + " ^9CA3AF" + .@i + "   ยังไม่ปลดล็อก (ใช้ Race Slot Key)^000000";
		} else if (!.@r) {
			mes .@mk$ + " ^6B7280" + .@i + "^000000   ^9CA3AF- ว่าง -^000000" + ((race_lock & (1 << .@i)) ? " ^B91C1C[ล็อก]^000000" : "");
		} else {
			.@l$ = .@mk$ + " ^6B7280" + .@i + "^000000 " + F_RaceData(.@r, 2) + "|||^000000 " + F_RaceName(.@r) + "  " + F_RaceStarBar(getd("race_st" + .@r)) + "  " + F_RaceRarity(F_RaceData(.@r, 5), 1) + F_RaceRarity(F_RaceData(.@r, 5), 0) + "^000000";
			if (race_lock & (1 << .@i))
				.@l$ += "  ^B91C1C[ล็อก]^000000";
			mes .@l$;
		}
	}
	return;

L_Msg:
	callsub L_Title, "ระบบเผ่า";
	mes getarg(0);
	next;
	return;

L_Clr:
	mes " ";
	clear;
	return;

L_Title:
	mes " ";
	setdialogsize 480, 400;
	clear;
	setdialogalign DIALOG_ALIGN_LEFT;
	mes "^4C1D95========================================^000000";
	mes "^7E22CE RACE SYSTEM ^9CA3AF |^000000 ^0F766E" + getarg(0) + "^000000";
	mes "^4C1D95========================================^000000";
	return;
}

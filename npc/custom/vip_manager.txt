// Cash Shop (pays with #CASHPOINTS). Prices: see doc/rae/systems/cash_shop.md
// VIP Ticket 1/5/15/30 days, Auto Battle Pass Box 1/3/7 days
-	cashshop	RAE_CashShop	-1,32510:19,32511:79,32512:199,32513:299,32520:15,32521:39,32522:79

// VIP status text. getarg(0) = NPC name shown in the header
function	script	F_VIPStatus	{
	mes "^1E3A8A[ " + getarg(0) + " ]^000000";
	if (vip_status(VIP_STATUS_ACTIVE)) {
		.@remain = vip_status(VIP_STATUS_REMAINING);
		mes "สถานะ VIP: ^15803Dใช้งานอยู่^000000";
		mes "หมดอายุ: ^1D4ED8" + gettimestr("%d/%m/%Y %H:%M", 21, vip_status(VIP_STATUS_EXPIRE)) + "^000000";
		mes "เหลือเวลา: " + (.@remain / 86400) + " วัน " + ((.@remain % 86400) / 3600) + " ชม. " + ((.@remain % 3600) / 60) + " นาที";
	} else {
		mes "สถานะ VIP: ^B91C1Cยังไม่เป็น VIP^000000";
		mes "ใช้ ^1D4ED8VIP Ticket^000000 เพื่อเป็น VIP";
	}
	return;
}

// Item script of the VIP Tickets: F_VIPTicket(<days>)
// Item scripts cannot open a dialog, so everything is told in the chat box.
function	script	F_VIPTicket	{
	.@days = getarg(0);
	vip_time .@days * 1440; // minutes, added to the time that is left
	dispbottom "[VIP] เพิ่มเวลา VIP +" + .@days + " วัน เรียบร้อย", 0x66FF66;
	dispbottom "[VIP] หากเพิ่งเป็น VIP ครั้งแรก กรุณาออกจากเกมแล้วเข้าใหม่ เพื่อรับสิทธิ์คลังเก็บของ/ช่องตัวละครเพิ่ม", 0xFFCC66;
	end;
}

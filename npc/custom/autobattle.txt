//===== rAthena Script =======================================
//= Auto Battle - setup dialog (Thai)
//===== Description: =========================================
//= NPC based front-end for the server side Auto Battle system
//= (src/map/autobattle.cpp, command: @autobattle / @ab).
//= Players open it with the town NPC below, or by typing @ab anywhere.
//= Tabs: Hunt / Recovery / Buffs / Support / Loot / Restock / Safety
//===== Requires: ============================================
//= - Auto Battle patch applied (autobattle.cpp)
//= - Players need the "autobattle" permission (conf/groups.yml)
//===== Notes: ===============================================
//= ** THIS FILE MUST BE SAVED AS TIS-620 / cp874 (Thai) ** to be
//= readable by the Thai client.
//= The dialog reads / writes the settings only through the script
//= commands autobattleinfo(), autobattlestr() and autobattlecmd().
//= Change the coordinates of the town NPC to suit your map.
//============================================================

-	script	AutoBattleUI	-1,{
	end;

OnInit:
	// 1 = the NPC hands out the Auto Battle Pass (item 32500) when a player has none, 0 = never
	.give_pass = 1;
	end;

OnOpen:
	callfunc "F_ABUI";
	end;

OnOpenItem:
	// started by the Auto Battle item (see item 32500): the item itself counts as the pass
	callfunc "F_ABUI", 1;
	end;
}

prontera,155,181,4	script	Auto Battle Manager	4_F_KAFRA1,{
	callfunc "F_ABUI";
	end;
}

// Main menu. getarg(0) = 1 when opened by using the Auto Battle Pass item (start is always allowed then)
function	script	F_ABUI	{
	.@from_item = getarg(0, 0);
	if (!autobattleinfo("ready")) {
		callsub L_Title, "ข้อมูลยังไม่พร้อม";
		mes "ข้อมูลตัวละครยังโหลดไม่เสร็จ";
		mes "กรุณารอสักครู่แล้วลองใหม่อีกครั้ง";
		close;
	}

	while (1) {
		callsub L_Clr;
		.@run = autobattleinfo("running");
		.@item_only = getbattleflag("autobattle_item_only");
		.@can_start = (!.@item_only || .@from_item || countitem(32500) > 0);
		callsub L_Title, "ล่าอัตโนมัติ - เมนูหลัก";
		mes "สถานะ: " + (.@run ? "^15803D[ กำลังทำงาน ]^000000" : "^B91C1C[ หยุดอยู่ ]^000000");
		mes " ";
		callsub L_Dashboard;
		if (.@run)
			.@first$ = "^B91C1C>> หยุด Auto Battle<<^000000";
		else if (.@can_start)
			.@first$ = "^15803D>> เริ่ม Auto Battle <<^000000";
		else
			.@first$ = "^B45309>> วิธีเริ่มใช้งาน (ต้องมีไอเทม) <<^000000";
		.@s = select(.@first$ + ":^1E3A8A1.^000000 ล่ามอนสเตอร์ / สกิลโจมตี:^1E3A8A2.^000000 ฟื้นฟู HP-SP / นั่งพัก:^1E3A8A3.^000000 บัพตัวเอง:^1E3A8A4.^000000 ซัพพอร์ต / ฮีล:^1E3A8A5.^000000 เก็บของ:^1E3A8A6.^000000 ซื้อของอัตโนมัติ:^1E3A8A7.^000000 ความปลอดภัย:ดูสรุปในช่องแชท:รีเซ็ตการตั้งค่าทั้งหมด:ปิดหน้าต่าง");
		switch (.@s) {
		case 1:
			if (.@run) {
				autobattlecmd("stop");
			} else if (.@can_start) {
				autobattle 1;
				if (autobattleinfo("running"))
					callsub L_Bye, "เริ่ม Auto Battle แล้ว";
				callsub L_Clr;
				callsub L_Title, "เริ่มไม่สำเร็จ";
				mes "ดูสาเหตุในช่องแชท";
				mes "(เช่น ยังไม่ได้ตั้งสกิล / เลือกเป้าหมาย)";
				next;
			} else {
				callsub L_HowToStart;
			}
			break;
		case 2: callsub L_Hunt; break;
		case 3: callsub L_Recovery; break;
		case 4: callsub L_Buffs; break;
		case 5: callsub L_Support; break;
		case 6: callsub L_Loot; break;
		case 7: callsub L_Restock; break;
		case 8: callsub L_Safety; break;
		case 9: autobattlecmd("status"); break;
		case 10:
			callsub L_Clr;
			callsub L_Title, "รีเซ็ตการตั้งค่า";
			mes "การตั้งค่าทั้งหมดของ Auto Battle";
			mes "(สกิล ยา มอนสเตอร์ ฯลฯ) จะถูกล้าง";
			mes "ต้องหยุด Auto Battle ก่อนจึงจะรีเซ็ตได้";
			if (select("^B91C1Cรีเซ็ตเลย^000000:ยกเลิก") == 1)
				callsub L_Cmd, "reset";
			break;
		case 11: callsub L_Bye, "ปิดเมนูแล้ว";
		}
	}
	close;

// ------------------------------------------------------------ helpers
// colored ON / OFF for menus
L_OO:
	return (getarg(0) ? "^15803D[ เปิด ]^000000" : "^B91C1C[ ปิด ]^000000");

// colored text (for dialog lines)
L_OC:
	return (getarg(0) ? "^15803Dเปิด^000000" : "^B91C1Cปิด^000000");

// "clear" only works while a mes is active (it is not after select / next), so write a blank line first
L_Clr:
	mes " ";
	clear;
	return;

// Closes the dialog safely (close needs a mes) with a short confirmation text
L_Bye:
	callsub L_Clr;
	mes "^1E3A8A== AUTO BATTLE ==^000000";
	mes "^0F766E" + getarg(0) + "^000000";
	close;

// Styled header of every screen. getarg(0) = title. Also enlarges the dialog window.
L_Title:
	// Resizing the window leaves the client's text position wrong, so: size -> blank mes + clear -> align,
	// and only then the real text (without this the lines are drawn over the header)
	setdialogsize 520, 440;
	callsub L_Clr;
	setdialogalign DIALOG_ALIGN_LEFT;
	mes "^1E3A8A== AUTO BATTLE ==^000000";
	mes "^0F766E" + getarg(0) + "^000000";
	mes "^9CA3AF------------------------------------------^000000";
	return;

L_ItemName:
	if (getarg(0) <= 0)
		return "-";
	return getitemname(getarg(0));

// Runs an @ab command; tells the player when it did not work
L_Cmd:
	if (autobattlecmd(getarg(0)))
		return;
	callsub L_Clr;
	callsub L_Title, "ตั้งค่าไม่สำเร็จ";
	mes "ดูรายละเอียดในช่องแชท";
	next;
	return;

// counts the used slots: getarg(0) = info prefix (atk/buff/sup), getarg(1) = number of slots
L_CountSlots:
	.@n = 0;
	for (.@i = 0; .@i < getarg(1); .@i++)
		if (autobattleinfo(getarg(0) + "_id", .@i))
			.@n++;
	return .@n;

L_Dashboard:
	.@mn = autobattleinfo("mobn");
	mes "^6B7280เป้าหมาย^000000: " + (autobattleinfo("all") ? "ทุกตัวในแผนที่" : (.@mn ? .@mn + " ชนิดที่เลือก" : "^B91C1Cยังไม่ได้เลือก^000000")) + (autobattleinfo("fightback") ? " (สู้กลับ)" : "");
	mes "^6B7280โจมตี^000000: ธรรมดา " + callsub(L_OC, autobattleinfo("normal")) + " / สกิล " + callsub(L_CountSlots, "atk", 16) + " ช่อง";
	mes "^6B7280ยา HP^000000: " + callsub(L_ItemName, autobattleinfo("hpitem")) + " (<" + autobattleinfo("hppct") + "%)";
	mes "^6B7280ยา SP^000000: " + callsub(L_ItemName, autobattleinfo("spitem")) + " (<" + autobattleinfo("sppct") + "%)";
	mes "^6B7280บัพ^000000: " + callsub(L_CountSlots, "buff", 8) + " สกิล / ซัพพอร์ต: " + callsub(L_CountSlots, "sup", 4) + " สกิล";
	mes "^6B7280เก็บของ^000000: " + callsub(L_OC, autobattleinfo("loot")) + " / ซื้อของ: " + callsub(L_OC, autobattleinfo("restock"));
	mes "^6B7280หนีเมื่อ HP ต่ำกว่า^000000: " + (autobattleinfo("fleepct") ? autobattleinfo("fleepct") + "%" : "ปิด");
	mes " ";
	return;

L_HowToStart:
	callsub L_Clr;
	callsub L_Title, "วิธีเริ่มใช้งาน";
	mes "เซิร์ฟเวอร์นี้ใช้ไอเทม ^1D4ED8Auto Battle Pass^000000";
	mes "กดใช้ไอเทมเพื่อเปิดเมนูตั้งค่านี้";
	mes "แล้วกดปุ่ม เริ่ม / หยุด ในเมนูได้เลย";
	mes "ระหว่างทำงานจะมีไอคอนสถานะและเอฟเฟกต์";
	mes "ให้ผู้เล่นคนอื่นเห็นว่าใช้ Auto Battle อยู่";
	if (countitem(32500)) {
		mes " ";
		mes "คุณมีไอเทมนี้อยู่ในกระเป๋าแล้ว";
		next;
		return;
	}
	if (getvariableofnpc(.give_pass, "AutoBattleUI")) {
		if (select("รับไอเทม Auto Battle Pass:ย้อนกลับ") == 1) {
			getitem 32500, 1;
			mes "ได้รับ Auto Battle Pass แล้ว";
			next;
		}
		return;
	}
	next;
	return;

// ------------------------------------------------------------ generic pickers
// Pick one entry of @abm$[0..@abmn-1] (10 per page). getarg(0) = title. Returns the index, or -1 on cancel.
L_Pick:
	.@page = 0;
	while (1) {
		.@from = .@page * 10;
		.@to = .@from + 10;
		if (.@to > @abmn)
			.@to = @abmn;
		.@m$ = "";
		for (.@i = .@from; .@i < .@to; .@i++)
			.@m$ += @abm$[.@i] + ":";
		if (.@from > 0)
			.@m$ += "<< หน้าก่อนหน้า:";
		if (.@to < @abmn)
			.@m$ += ">> หน้าถัดไป:";
		.@m$ += "ยกเลิก";
		callsub L_Clr;
		callsub L_Title, "" + getarg(0) + "";
		mes "หน้า " + (.@page + 1) + " / " + ((@abmn + 9) / 10);
		.@c = select(.@m$) - 1;
		if (.@c < .@to - .@from)
			return .@from + .@c;
		.@k = .@to - .@from;
		if (.@from > 0) {
			if (.@c == .@k) {
				.@page--;
				continue;
			}
			.@k++;
		}
		if (.@to < @abmn) {
			if (.@c == .@k) {
				.@page++;
				continue;
			}
			.@k++;
		}
		return -1;
	}

// Pick an item. getarg(0): 0 = healing items, 1 = healing + usable, 2 = any stackable item.
// Returns the item ID (as text) or the typed name, "" when cancelled.
L_PickItem:
	.@mode = getarg(0);
	while (1) {
		callsub L_Clr;
		callsub L_Title, "เลือกไอเทม";
		mes "เลือกจากกระเป๋า หรือพิมพ์ ID / ชื่อเอง";
		.@c = select("เลือกจากไอเทมในกระเป๋า:พิมพ์ ID หรือชื่อไอเทม:ยกเลิก");
		if (.@c == 3)
			return "";
		if (.@c == 2) {
			callsub L_Clr;
			mes "พิมพ์ ID หรือชื่อไอเทม (ชื่อแบบ Aegis)";
			mes "เช่น ^1D4ED8501^000000 หรือ ^1D4ED8Red_Potion^000000";
			input .@s$;
			.@s$ = replacestr(.@s$, " ", "_");
			if (.@s$ != "")
				return .@s$;
			continue;
		}
		// the list is built by the server (a script loop over the whole inventory hits the loop limit)
		autobattlecmd("items " + .@mode);
		.@n = autobattleinfo("itn");
		deletearray @abm$[0], getarraysize(@abm$);
		deletearray @abid[0], getarraysize(@abid);
		for (.@i = 0; .@i < .@n; .@i++) {
			@abid[.@i] = autobattleinfo("itid", .@i);
			@abm$[.@i] = autobattlestr("itname", .@i) + " (" + autobattleinfo("itcnt", .@i) + ")";
		}
		@abmn = .@n;
		if (!.@n) {
			callsub L_Clr;
			mes "ไม่พบไอเทมที่เหมาะสมในกระเป๋า";
			mes "ลองเลือก 'พิมพ์ ID หรือชื่อ' แทน";
			next;
			continue;
		}
		.@p = callsub(L_Pick, "เลือกไอเทม");
		if (.@p >= 0)
			return "" + @abid[.@p];
	}

// Pick a learned skill. getarg(0) = attack | buff | support. Returns the skill ID (0 = cancel), level in @ablv.
L_PickSkill:
	autobattlecmd("skills " + getarg(0));
	.@n = autobattleinfo("skn");
	if (!.@n) {
		callsub L_Clr;
		callsub L_Title, "เลือกสกิล";
		mes "คุณยังไม่มีสกิลที่ใช้ได้ในหมวดนี้";
		mes "(ต้องเรียนสกิลก่อน)";
		next;
		return 0;
	}
	deletearray @abm$[0], getarraysize(@abm$);
	for (.@i = 0; .@i < .@n; .@i++)
		@abm$[.@i] = autobattlestr("skname", .@i) + " (Lv " + autobattleinfo("sklv", .@i) + ")";
	@abmn = .@n;
	.@p = callsub(L_Pick, "เลือกสกิล");
	if (.@p < 0)
		return 0;
	.@id = autobattleinfo("skid", .@p);
	.@max = autobattleinfo("sklv", .@p);
	@ablv = .@max;
	if (.@max > 1) {
		callsub L_Clr;
		mes "^1D4ED8" + autobattlestr("skill", .@id) + "^000000";
		mes "คุณเรียนไว้ Lv " + .@max;
		if (select("ใช้เลเวลสูงสุด (Lv " + .@max + "):เลือกเลเวลเอง") == 2) {
			mes "พิมพ์เลเวลที่ต้องการ (1-" + .@max + ")";
			input .@lv, 1, .@max;
			@ablv = .@lv;
		}
	}
	return .@id;

// re-applies the HP condition of an attack skill: getarg(0) = slot (1-based), getarg(1) = cond, getarg(2) = value
L_SetCond:
	if (getarg(1) == 1)
		autobattlecmd("skill cond " + getarg(0) + " above " + getarg(2));
	else if (getarg(1) == 2)
		autobattlecmd("skill cond " + getarg(0) + " below " + getarg(2));
	return;

// Skill list manager.
// getarg: 0 = title, 1 = kind (attack/buff/support), 2 = command (skill/buff/support),
//         3 = info prefix (atk/buff/sup), 4 = number of slots
L_Skills:
	.@title$ = getarg(0);
	.@kind$ = getarg(1);
	.@cmd$ = getarg(2);
	.@ip$ = getarg(3);
	.@n = getarg(4);
	.@isatk = (.@kind$ == "attack");
	while (1) {
		callsub L_Clr;
		callsub L_Title, "" + .@title$ + "";
		deletearray @abs[0], getarraysize(@abs);
		.@cnt = 0;
		.@m$ = "";
		for (.@i = 0; .@i < .@n; .@i++) {
			.@id = autobattleinfo(.@ip$ + "_id", .@i);
			if (!.@id)
				continue;
			@abs[.@cnt] = .@i;
			.@line$ = autobattlestr(.@ip$, .@i) + " Lv" + autobattleinfo(.@ip$ + "_lv", .@i);
			if (.@isatk) {
				.@cd = autobattleinfo("atk_cond", .@i);
				if (.@cd == 1)
					.@line$ += " [HP มอน >" + autobattleinfo("atk_cv", .@i) + "%]";
				else if (.@cd == 2)
					.@line$ += " [HP มอน <" + autobattleinfo("atk_cv", .@i) + "%]";
			}
			.@cnt++;
			mes .@cnt + ". " + .@line$;
			.@m$ += .@cnt + ". " + .@line$ + ":";
		}
		if (!.@cnt)
			mes "(ยังไม่มีสกิลในรายการ)";
		mes " ";
		if (.@isatk)
			mes "สกิลบนสุดจะถูกใช้ก่อน ถ้าใช้ไม่ได้จะลองสกิลถัดไป";
		else if (.@kind$ == "buff")
			mes "ระบบจะใช้ซ้ำอัตโนมัติเมื่อบัพหมดเวลา";
		else
			mes "ใช้กับตัวเองและเพื่อนในปาร์ตี้ที่ HP ต่ำ";
		mes "เลือกสกิลในรายการเพื่อแก้ไข / ลบ";
		.@m$ += "+ เพิ่มสกิล:ล้างสกิลทั้งหมด:ย้อนกลับ";
		.@c = select(.@m$);

		if (.@c <= .@cnt) {
			// ---- edit one entry
			.@slot = @abs[.@c - 1];
			.@k = 0;
			.@e$ = "ลบสกิลนี้:";
			.@act[.@k] = 1;
			.@k++;
			if (.@c > 1) {
				.@e$ += "ย้ายขึ้นบน (ใช้ก่อน):";
				.@act[.@k] = 2;
				.@k++;
			}
			if (.@isatk) {
				.@e$ += "ตั้งเงื่อนไข HP ของมอนสเตอร์:";
				.@act[.@k] = 3;
				.@k++;
			}
			.@e$ += "เปลี่ยนเลเวล:";
			.@act[.@k] = 4;
			.@k++;
			.@e$ += "ย้อนกลับ";
			.@act[.@k] = 0;
			callsub L_Clr;
			mes "^1D4ED8" + autobattlestr(.@ip$, .@slot) + " Lv" + autobattleinfo(.@ip$ + "_lv", .@slot) + "^000000";
			.@a = .@act[select(.@e$) - 1];
			if (.@a == 1) {
				callsub L_Cmd, .@cmd$ + " clear " + (.@slot + 1);
			} else if (.@a == 2) {
				.@prev = @abs[.@c - 2];
				.@i1 = autobattleinfo(.@ip$ + "_id", .@slot);
				.@l1 = autobattleinfo(.@ip$ + "_lv", .@slot);
				.@i2 = autobattleinfo(.@ip$ + "_id", .@prev);
				.@l2 = autobattleinfo(.@ip$ + "_lv", .@prev);
				if (.@isatk) {
					.@c1 = autobattleinfo("atk_cond", .@slot);
					.@v1 = autobattleinfo("atk_cv", .@slot);
					.@c2 = autobattleinfo("atk_cond", .@prev);
					.@v2 = autobattleinfo("atk_cv", .@prev);
				}
				autobattlecmd(.@cmd$ + " " + (.@prev + 1) + " " + .@i1 + " " + .@l1);
				autobattlecmd(.@cmd$ + " " + (.@slot + 1) + " " + .@i2 + " " + .@l2);
				if (.@isatk) {
					callsub L_SetCond, (.@prev + 1), .@c1, .@v1;
					callsub L_SetCond, (.@slot + 1), .@c2, .@v2;
				}
			} else if (.@a == 3) {
				callsub L_Clr;
				callsub L_Title, "เงื่อนไขการใช้สกิล";
				mes autobattlestr("atk", .@slot);
				.@w = select("ใช้ได้ทุกกรณี:ใช้เมื่อ HP มอนสเตอร์ มากกว่า N%:ใช้เมื่อ HP มอนสเตอร์ น้อยกว่า N%:ย้อนกลับ");
				if (.@w == 1) {
					callsub L_Cmd, "skill cond " + (.@slot + 1) + " always";
				} else if (.@w == 2 || .@w == 3) {
					mes "พิมพ์ค่า N (1-99)";
					input .@v, 1, 99;
					callsub L_Cmd, "skill cond " + (.@slot + 1) + (.@w == 2 ? " above " : " below ") + .@v;
				}
			} else if (.@a == 4) {
				.@sid = autobattleinfo(.@ip$ + "_id", .@slot);
				.@max = autobattleinfo("learned", .@sid);
				if (.@max > 1) {
					mes "พิมพ์เลเวลที่ต้องการ (1-" + .@max + ")";
					input .@lv, 1, .@max;
					if (.@isatk) {
						.@c1 = autobattleinfo("atk_cond", .@slot);
						.@v1 = autobattleinfo("atk_cv", .@slot);
					}
					callsub L_Cmd, .@cmd$ + " " + (.@slot + 1) + " " + .@sid + " " + .@lv;
					if (.@isatk)
						callsub L_SetCond, (.@slot + 1), .@c1, .@v1;
				} else {
					mes "สกิลนี้เรียนไว้ Lv 1 เท่านั้น";
					next;
				}
			}
		} else if (.@c == .@cnt + 1) {
			// ---- add a skill in the first free slot
			.@free = -1;
			for (.@i = 0; .@i < .@n; .@i++) {
				if (!autobattleinfo(.@ip$ + "_id", .@i)) {
					.@free = .@i;
					break;
				}
			}

			callsub L_Clr;
			mes "หนีเมื่อ HP ต่ำกว่ากี่ %";
			mes "พิมพ์ 0-90 (0 = ปิด)";
			input .@v, 0, 90;
			callsub L_Cmd, "safety hp " + .@v;
			break;
		case 2:
			.@it$ = callsub(L_PickItem, 2);
			if (.@it$ != "")
				callsub L_Cmd, "safety item " + .@it$;
			break;
		case 3: callsub L_Cmd, "safety item off"; break;
		case 4: callsub L_Cmd, "safety player " + (autobattleinfo("stopplayer") ? "off" : "on"); break;
		case 5: callsub L_Cmd, "safety weight " + (autobattleinfo("stopweight") ? "off" : "on"); break;
		case 6:
			callsub L_Clr;
			mes "หยุดเมื่อน้ำหนักถึงกี่ %";
			mes "พิมพ์ 10-100";
			input .@v, 10, 100;
			callsub L_Cmd, "safety weight on " + .@v;
			break;
		case 7: callsub L_Cmd, "safety weightwarp " + (autobattleinfo("weightwarp") ? "off" : "on"); break;
		case 8: callsub L_Cmd, "safety noitem " + (autobattleinfo("stopnoitem") ? "off" : "on"); break;
		case 9: callsub L_Cmd, "safety mvp " + (autobattleinfo("fleemvp") ? "off" : "on"); break;
		case 10: callsub L_Cmd, "safety mini " + (autobattleinfo("fleemini") ? "off" : "on"); break;
		case 11:
			callsub L_Clr;
			mes "ไม่เจอมอนสเตอร์นานกี่วินาที";
			mes "ถึงจะใช้ไอเทมหนีเพื่อย้ายจุดล่า";
			mes "พิมพ์ 0-3600 (0 = ปิด)";
			input .@v, 0, 3600;
			callsub L_Cmd, "safety idle " + .@v;
			break;
		case 12: return;
		}
	}
